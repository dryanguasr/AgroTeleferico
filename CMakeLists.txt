cmake_minimum_required(VERSION 3.16)
project(agro_teleferico)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Bind calibration to the shared source and compiler, in CLI, plugin and tests.
set(AGRO_CORE_INPUT "${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}-${CMAKE_CXX_FLAGS}-${CMAKE_BUILD_TYPE}")
foreach(source include/agro/loop.hpp include/agro/demand.hpp include/agro/sha256.hpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" digest)
  string(APPEND AGRO_CORE_INPUT "-${digest}")
endforeach()
string(SHA256 AGRO_CORE_ID "${AGRO_CORE_INPUT}")
add_compile_definitions(AGRO_CORE_ID="${AGRO_CORE_ID}")
option(AGRO_LOOP_STANDALONE "Build the reduced loop model without ROS/Gazebo" OFF)
include(CTest)
add_executable(agro_loop_cli src/loop_cli.cpp)
target_include_directories(agro_loop_cli PRIVATE include)
target_compile_options(agro_loop_cli PRIVATE -Wall -Wextra -O2)
if(BUILD_TESTING)

  target_include_directories(test_ballasted PRIVATE include)

  target_include_directories(test_loop PRIVATE include)
  target_compile_options(test_loop PRIVATE -Wall -Wextra -O2)
  add_test(NAME loop_core COMMAND test_loop)

  target_include_directories(test_demand PRIVATE include)
  target_compile_options(test_demand PRIVATE -Wall -Wextra -O2)
  add_test(NAME demand_core COMMAND test_demand)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME ballasted_assets COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ballasted_assets.py)
  add_test(NAME ballasted_geometry COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ballasted_geometry.py)
  add_test(NAME loop_assets COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_loop_assets.py)
  add_test(NAME loop_guard COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_loop_guard.py)
  add_test(NAME ros_interface_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ros_interface.py)

  add_test(NAME demand_energy_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_demand_energy.py)
  add_test(NAME demand_smoke_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_demand_smoke.py)
endif()
if(AGRO_LOOP_STANDALONE)
  message(STATUS "Standalone reduced model only: Gazebo/ROS validation is not part of this build")
  return()
endif()
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_library(AgroCableSystem SHARED src/system.cpp)
target_include_directories(AgroCableSystem PRIVATE include)
target_link_libraries(AgroCableSystem gz-sim8::gz-sim8 gz-plugin2::register Eigen3::Eigen)
target_compile_options(AgroCableSystem PRIVATE -Wall -Wextra -O2)
add_library(AgroLoopSystem SHARED src/loop_system.cpp)
target_include_directories(AgroLoopSystem PRIVATE include)
target_link_libraries(AgroLoopSystem gz-sim8::gz-sim8 gz-plugin2::register)
target_compile_options(AgroLoopSystem PRIVATE -Wall -Wextra -O2)
install(TARGETS AgroCableSystem AgroLoopSystem DESTINATION lib)
install(TARGETS agro_loop_cli DESTINATION lib/${PROJECT_NAME})
add_executable(agro_telemetry_bridge src/telemetry_bridge.cpp)
target_link_libraries(agro_telemetry_bridge gz-sim8::gz-sim8)
ament_target_dependencies(agro_telemetry_bridge rclcpp std_msgs)
install(TARGETS agro_telemetry_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/agro DESTINATION share/${PROJECT_NAME}/include)
execute_process(COMMAND git describe --always --dirty
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE AGRO_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/revision.txt "${AGRO_REVISION}\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/revision.txt DESTINATION share/${PROJECT_NAME})

install(PROGRAMS scripts/validate_demand_native.py scripts/compose_telemetry_hud.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/run_demand_smoke.py scripts/audit_demand_outputs.py scripts/audit_demand_energy.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_physics tests/test_physics.cpp)
