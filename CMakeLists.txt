cmake_minimum_required(VERSION 3.16)
project(agro_teleferico)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AGRO_LOOP_STANDALONE "Build the reduced loop model without ROS/Gazebo" OFF)
include(CTest)
add_executable(agro_loop_cli src/loop_cli.cpp)
target_include_directories(agro_loop_cli PRIVATE include)
target_compile_options(agro_loop_cli PRIVATE -Wall -Wextra -O2)
if(BUILD_TESTING)
  add_executable(test_loop tests/test_loop.cpp)
  target_include_directories(test_loop PRIVATE include)
  target_compile_options(test_loop PRIVATE -Wall -Wextra -O2)
  add_test(NAME loop_core COMMAND test_loop)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME loop_assets COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_loop_assets.py)
  add_test(NAME loop_guard COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_loop_guard.py)
endif()
if(AGRO_LOOP_STANDALONE)
  message(STATUS "Standalone reduced model only: Gazebo/ROS validation is not part of this build")
  return()
endif()
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_library(AgroCableSystem SHARED src/system.cpp)
target_include_directories(AgroCableSystem PRIVATE include)
target_link_libraries(AgroCableSystem gz-sim8::gz-sim8 gz-plugin2::register Eigen3::Eigen)
target_compile_options(AgroCableSystem PRIVATE -Wall -Wextra -O2)
add_library(AgroLoopSystem SHARED src/loop_system.cpp)
target_include_directories(AgroLoopSystem PRIVATE include)
target_link_libraries(AgroLoopSystem gz-sim8::gz-sim8 gz-plugin2::register)
target_compile_options(AgroLoopSystem PRIVATE -Wall -Wextra -O2)
install(TARGETS AgroCableSystem AgroLoopSystem DESTINATION lib)
install(TARGETS agro_loop_cli DESTINATION lib/${PROJECT_NAME})
add_executable(agro_telemetry_bridge src/telemetry_bridge.cpp)
target_link_libraries(agro_telemetry_bridge gz-sim8::gz-sim8)
ament_target_dependencies(agro_telemetry_bridge rclcpp std_msgs)
install(TARGETS agro_telemetry_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/generate_world.py scripts/ros_interface.py scripts/run_suite.py scripts/render_replay.py scripts/capture_scene.py scripts/capture_loop.py scripts/generate_loop.py scripts/run_loop_suite.py scripts/report_loop.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/run_loop_guarded.py scripts/compare_loop_backends.py scripts/collect_loop_native.py scripts/plot_loop_native.py scripts/verify_loop_capture.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(test_physics tests/test_physics.cpp)
  target_include_directories(test_physics PRIVATE include)
  target_link_libraries(test_physics Eigen3::Eigen)
  add_test(NAME physics COMMAND test_physics)
endif()
ament_package()
