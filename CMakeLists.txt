cmake_minimum_required(VERSION 3.16)
project(agro_teleferico)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_library(AgroCableSystem SHARED src/system.cpp)
target_include_directories(AgroCableSystem PRIVATE include)
target_link_libraries(AgroCableSystem gz-sim8::gz-sim8 gz-plugin2::register Eigen3::Eigen)
target_compile_options(AgroCableSystem PRIVATE -Wall -Wextra -O2)
install(TARGETS AgroCableSystem DESTINATION lib)
add_executable(agro_telemetry_bridge src/telemetry_bridge.cpp)
target_link_libraries(agro_telemetry_bridge gz-sim8::gz-sim8)
ament_target_dependencies(agro_telemetry_bridge rclcpp std_msgs)
install(TARGETS agro_telemetry_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/generate_world.py scripts/ros_interface.py scripts/run_suite.py scripts/render_replay.py scripts/capture_scene.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_physics tests/test_physics.cpp)
  target_include_directories(test_physics PRIVATE include)
  target_link_libraries(test_physics Eigen3::Eigen)
  add_test(NAME physics COMMAND test_physics)
endif()
ament_package()
